\chapter{Código Fuente de \texttt{CodeT5InputBuilder.ts}}
\label{anx:codigo-builder}

Este anexo presenta la implementación completa del componente \texttt{CodeT5InputBuilder.ts}, encargado de recorrer el árbol sintáctico generado por ANTLR y construir el modelo semántico intermedio (\texttt{SemanticModel}) que alimenta tanto la capa determinística como el motor neuronal CodeT5.

\begin{lstlisting}[language=Java, caption={Implementación del constructor semántico en TypeScript}, label={lst:code-t5-builder}, numbers=left, breaklines=true, basicstyle=\ttfamily\small]
import idiomsListener from "../../antlr/idiomsListener";

import {
    AttrContext,
    ClasificationContext,
    CompositionContext,
    EntitytypeContext,
    Historic_reflexiveContext,
    IdentifierContext,
    Is_aContext,
    MasterdetailContext,
    ReflexiveContext
} from "../../antlr/idiomsParser";

// Interfaces del modelo
export interface Attribute {
    name: string;
    type: string;
}
export interface Identifier {
    name: string;
    type: string;
}
export interface Entity {
    name: string;
    identifier?: Identifier;
    attributes: Attribute[];
}
export interface Relation {
    type: string;
    from: string;
    to: string;
    role?: string;
    roles?: string[];
}
export interface SemanticModel {
    language: string;
    entities: Entity[];
    relations: Relation[];
}

// Builder
export default class CodeT5InputBuilder extends idiomsListener {
    semantic: SemanticModel;
    currentEntity: Entity | null;
    
    constructor() {
        super();
        this.semantic = {
            language: "",
            entities: [],
            relations: []
        };
        this.currentEntity = null;
    }

    public setLanguage(language: string): void {
        this.semantic.language = language;
    }

    /* ENTIDADES */
    enterEntitytype = (ctx: EntitytypeContext) => {
        const entity: Entity = {
            name: ctx.ID().getText(),
            attributes: []
        };
        this.semantic.entities.push(entity);
        this.currentEntity = entity;
    }

    enterIdentifier = (ctx: IdentifierContext) => {
        if (!this.currentEntity) return;
        this.currentEntity.identifier = {
            name: ctx.FIELDNAME().getText(),
            type: ctx.TYPE().getText()
        };
    }

    enterAttr = (ctx: AttrContext) => {
        if (!this.currentEntity) return;
        this.currentEntity.attributes.push({
            name: ctx.FIELDNAME().getText(),
            type: ctx.TYPE().getText()
        });
    }

    /* RELACIONES */
    private cleanRole(role?: string): string | undefined {
        return role?.replace(/"/g, "");
    }

    enterClasification = (ctx: ClasificationContext) => {
        this.semantic.relations.push({
            type: "classification",
            from: ctx.classified().entity().ID().getText(),
            to: ctx.classifier().entity().ID().getText(),
            role: this.cleanRole(
                ctx.classifier().entity().ROLE()?.getText() ??
                ctx.classified().entity().ROLE()?.getText()
            )
        });
    }

    enterMasterdetail = (ctx: MasterdetailContext) => {
        this.semantic.relations.push({
            type: "master-detail",
            from: ctx.master().entity().ID().getText(),
            to: ctx.detail().entity().ID().getText(),
            role: this.cleanRole(ctx.detail().entity().ROLE()?.getText())
        });
    }

    enterComposition = (ctx: CompositionContext) => {
        const composed = ctx.composed().entity().ID().getText();
        ctx.composer_list().forEach(composer => {
            this.semantic.relations.push({
                type: "composition",
                from: composer.entity().ID().getText(),
                to: composed,
                role: this.cleanRole(composer.entity().ROLE()?.getText())
            });
        });
    }

    enterIs_a = (ctx: Is_aContext) => {
        this.semantic.relations.push({
            type: "is-a",
            from: ctx.supertype().entity().ID().getText(),
            to: ctx.subtype().entity().ID().getText()
        });
    }

    enterHistoric_reflexive = (ctx: Historic_reflexiveContext) => {
        const roles = [
            ctx.reflexive_type().entity().ROLE()?.getText(),
            ctx.ROLE()?.getText(),
        ].map(r => this.cleanRole(r)).filter((r): r is string => !!r);

        this.semantic.relations.push({
            type: "historic-reflexive",
            from: ctx.reflexive_type().entity().ID().getText(),
            to: ctx.history().entity().ID().getText(),
            roles,
        });
    };

    enterReflexive = (ctx: ReflexiveContext) => {
        const entity = ctx.entity();
        const name = entity.ID().getText();

        this.semantic.relations.push({
            type: "reflexive",
            from: name,
            to: name,
            role: this.cleanRole(entity.ROLE()?.getText()),
        });
    };

    /* Utilidades */
    getSemanticModel(): SemanticModel {
        return this.semantic;
    }

    toJSON(): string {
        return JSON.stringify(this.semantic, null, 4);
    }
}
\end{lstlisting}

\chapter{Resultados de Pruebas Preliminares y Experimentos Negativos (Zero-Shot)}
\label{anx:prueba-zeroshot}

Como parte de la fase de experimentación inicial, se evaluó el comportamiento del modelo base \textit{Salesforce/codet5-base} bajo un esquema \textit{Zero-Shot} (es decir, sin aplicar el proceso de ajuste fino o \textit{fine-tuning} con el dataset específico del proyecto). 

El objetivo de esta prueba fue analizar la capacidad nativa del modelo preentrenado para interpretar la estructura de un diagrama conceptual y traducirla directamente a los campos de Django ORM.

\section{Salida Obtenida en Entorno Zero-Shot}
Al someter al modelo base a una inferencia directa utilizando el contrato estructurado, se observó una degradación severa en la generación, caracterizada por la aparición de bucles de atención y alucinaciones sintácticas. Un ejemplo representativo del resultado obtenido fue el siguiente:
\begin{lstlisting}[caption={Resultado de alucinación y bucle infinito en inferencia Zero-Shot con CodeT5-base}, label={anx:zeroshot-output}]
{
  "name": "User",
  "name": "Pais",
  "name": "User",
  "attributes": [
    {
      "name": "name",
      "type": "string"
      "name": "Pais",
      "name": "User",
      "name": "Pais",
      "name": "User",
      "name": "Pais",
      "name": "User",
      "name": "Pais"
    }
  ]}
\end{lstlisting}

\section{Análisis del Comportamiento}

Los resultados evidenciaron que el modelo base:
\begin{enumerate}
    \item \textbf{Carece de alineación de instrucciones:} Al no haber sido entrenado con el prompt específico del dominio de modelado conceptual, repite nombres de entidades y atributos de manera arbitraria.
    \item \textbf{sufre de bucles de generación (\textit{Repetition Loops}):} Entra en redundancias sintácticas al intentar completar el JSON, rompiendo por completo la estructura esperada para el analizador posterior.
\end{enumerate}

Este comportamiento adverso justificó la necesidad imperativa de desarrollar un conjunto de datos especializado (Dataset v6) y aplicar un proceso riguroso de \textit{fine-tuning}.

\chapter{Scripts de Entrenamiento y Fine-Tuning}
\label{anx:script-entrenamiento}

En este anexo se presenta el script en Python utilizado para llevar a cabo el entrenamiento incremental y el ajuste fino (\textit{fine-tuning}) del modelo CodeT5, partiendo de versiones previas y utilizando el conjunto de datos estructurado.

\begin{lstlisting}[caption={Script principal para el fine-tuning e entrenamiento incremental de CodeT5 (train.py)}, label={anx:train-script}]
from datasets import load_dataset
from transformers import (
    AutoTokenizer,
    AutoModelForSeq2SeqLM,
    Seq2SeqTrainer,
    Seq2SeqTrainingArguments
)
import torch
import os


# CONFIGURACION
# modelo V4 ya entrenado.
MODEL_PATH = "../codet5-django-v4.1"

DATASET_PATH = (
    "../dataset/django_dataset_v42.jsonl"
)

OUTPUT_DIR = "../codet5-django-v4.2"

MAX_INPUT = 512
MAX_OUTPUT = 512

if not os.path.exists(MODEL_PATH):

    raise FileNotFoundError(
        f"No se encontro el modelo V4 en: {MODEL_PATH}"
    )


# Cargar Dataset
print("Loading classification dataset...")
dataset = load_dataset(
    "json",
    data_files={
        "train": DATASET_PATH
    }
)

print(dataset)
# Tokenizer
print("\nLoading tokenizer...")

tokenizer = AutoTokenizer.from_pretrained(
    MODEL_PATH
)

print("\nLoading V4 model...")

model = AutoModelForSeq2SeqLM.from_pretrained(
    MODEL_PATH
)

print("V4 model loaded successfully!")

print(
    "\nStarting incremental training from:"
)
print(
    MODEL_PATH
)


# Pre - Procesamiento
def preprocess(example):

    prompt = (
        example["instruction"]
        + "\n\n"
        + example["input"]
    )

    # Input
    model_inputs = tokenizer(
        prompt,
        max_length=MAX_INPUT,
        truncation=True,
        padding="max_length"
    )

    # Output
    labels = tokenizer(
        example["output"],
        max_length=MAX_OUTPUT,
        truncation=True,
        padding="max_length"
    )

    # Ignorar padding en loss
    labels_ids = labels["input_ids"]

    labels_ids = [
        token
        if token != tokenizer.pad_token_id
        else -100
        for token in labels_ids
    ]

    model_inputs["labels"] = labels_ids

    return model_inputs


# Tokenizer
print("\nTokenizing dataset...")
tokenized = dataset["train"].map(
    preprocess
)
print("Tokenization finished!")

# Training Arguments
training_args = Seq2SeqTrainingArguments(
    output_dir=OUTPUT_DIR,
    overwrite_output_dir=True,

    # Pocas epocas porque NO estamos entrenando desde cero.
    num_train_epochs=3,

    # Learning rate pequenio, no modificando lo aprendido previamente
    learning_rate=1e-5,

    per_device_train_batch_size=4,

    # Acumulamos gradientes para tener un batch efectivo mayor
    gradient_accumulation_steps=1,

    save_steps=100,
    save_total_limit=2,
    logging_steps=20,
    predict_with_generate=True,
    fp16=torch.cuda.is_available(),

    report_to="none"
)

# TRAINER
trainer = Seq2SeqTrainer(
    model=model,
    args=training_args,
    train_dataset=tokenized,
    tokenizer=tokenizer
)

# Entrenamiento
print("Starting V4.1 incremental training...")

trainer.train()
print("Saving V4.1 model...")

trainer.save_model(
    OUTPUT_DIR
)
tokenizer.save_pretrained(
    OUTPUT_DIR
)
print("Training finished successfully!")
print(
    "New model:",
    OUTPUT_DIR
)
\end{lstlisting}
